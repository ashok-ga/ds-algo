    REST API: Medical Records by Weight¶

    Problem Statement¶

    Use the HTTP GET method to retrieve information from a database of patient medical records. Query
    https://jsonmock.hackerrank.com/api/medical_records
    to find all the records. The query result is paginated and can be further accessed by appending to the query string ?&page=num where num is the page number.

    Each response contains a JSON with these fields: - page: current page - per_page: max results per page - total: total number of records in the search result - total_pages: total number of pages - data: array of JSON objects with medical records

    Each record contains: - id: unique record id - timestamp: when record was generated (ms UTC) - userId: user id - userName, userDob - vitals: with fields like breathingRate - diagnosis: object with fields id, name, severity - doctor: object with fields id, name - meta: object with height and weight

    Fetch all medical records matching:¶

    1. Records fetched should be diagnosed by the doctor with id doctorId and the diagnosis should belong to diagnosisId.
    2. For the users matching the first criteria:
    3. meta.weight should be greater than their personal average weight among the matched records
    4. vitals.breathingRate in the record should be greater than the given input breathingRate

    If no records match, return [-1]. Otherwise, return a sorted array of ids.

    Java Solution¶

    import java.io.*;
import java.net.*;
import java.util.*;
import org.json.*;

public class Solution {
    public static int[] getRecordsDiagnosed(int doctorId, int breathingRate, int diagnosisId) throws Exception {
        List<JSONObject> allRecords = new ArrayList<>();
        int page = 1, totalPages = 1;
        while (page <= totalPages) {
            String url = "https://jsonmock.hackerrank.com/api/medical_records?page=" + page;
            HttpURLConnection conn = (HttpURLConnection) new URL(url).openConnection();
            conn.setRequestMethod("GET");
            BufferedReader reader = new BufferedReader(new InputStreamReader(conn.getInputStream()));
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) sb.append(line);
            JSONObject obj = new JSONObject(sb.toString());
            totalPages = obj.getInt("total_pages");
            JSONArray data = obj.getJSONArray("data");
            for (int i = 0; i < data.length(); i++) allRecords.add(data.getJSONObject(i));
            page++;
        }

        Map<Integer, List<Integer>> userWeights = new HashMap<>();
        List<JSONObject> filtered = new ArrayList<>();
        for (JSONObject record : allRecords) {
            JSONObject diag = record.getJSONObject("diagnosis");
            JSONObject doc = record.getJSONObject("doctor");
            if (doc.getInt("id") == doctorId && diag.getInt("id") == diagnosisId) {
                filtered.add(record);
                int userId = record.getInt("userId");
                JSONObject meta = record.getJSONObject("meta");
                userWeights.putIfAbsent(userId, new ArrayList<>());
                userWeights.get(userId).add(meta.getInt("weight"));
            }
        }

        Map<Integer, Double> avgWeight = new HashMap<>();
        for (Map.Entry<Integer, List<Integer>> e : userWeights.entrySet()) {
            double sum = 0;
            for (int w : e.getValue()) sum += w;
            avgWeight.put(e.getKey(), sum / e.getValue().size());
        }

        List<Integer> answer = new ArrayList<>();
        for (JSONObject record : filtered) {
            int userId = record.getInt("userId");
            JSONObject meta = record.getJSONObject("meta");
            JSONObject vitals = record.getJSONObject("vitals");
            int weight = meta.getInt("weight");
            int br = vitals.getInt("breathingRate");
            if (weight > avgWeight.get(userId) && br > breathingRate) {
                answer.add(record.getInt("id"));
            }
        }

        if (answer.isEmpty()) return new int[]{-1};
        Collections.sort(answer);
        int[] arr = new int[answer.size()];
        for (int i = 0; i < answer.size(); i++) arr[i] = answer.get(i);
        return arr;
    }

    public static void main(String[] args) throws Exception {
        int doctorId = 3;
        int breathingRate = 20;
        int diagnosisId = 2;
        int[] result = getRecordsDiagnosed(doctorId, breathingRate, diagnosisId);
        for (int id : result) System.out.println(id);
    }
}
