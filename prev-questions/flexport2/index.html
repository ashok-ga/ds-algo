    🧠 Problem: Even-Odd Operations¶

    📝 Description¶

    Given an array of non-negative integers, perform a series of operations until the array becomes empty. The score is updated at each operation based on the sum of elements currently in the array. The goal is to maximize the overall score at the end.

    🔢 Operations¶

    • Operations are 1-indexed.
    • Odd-indexed operation: Add the sum of all integers in the array to the score.
    • Even-indexed operation: Subtract the sum of all integers in the array from the score.
    • After each operation, you must remove either the leftmost or rightmost element to continue.

    Return the maximum score possible.

    📥 Input¶

    • An array of integers integerArray with size n (1 ≤ n ≤ 1000)
    • Each element is in range 0 ≤ integerArray[i] ≤ 10⁹

    📤 Output¶

    • Return an integer: the maximum possible score

    📘 Example¶

    Input 

    integerArray = [1, 2, 3, 4, 2, 6]

    Output 

    13

    Explanation

    • Op 1 (odd): score += 18 → score = 18 → remove 6 → arr = [1,2,3,4,2]
    • Op 2 (even): score -= 12 → score = 6 → remove 2 → arr = [1,2,3,4]
    • Op 3 (odd): score += 10 → score = 16 → remove 4 → arr = [1,2,3]
    • Op 4 (even): score -= 6 → score = 10 → remove 1 → arr = [2,3]
    • Op 5 (odd): score += 5 → score = 15 → remove 3 → arr = [2]
    • Op 6 (even): score -= 2 → score = 13 → arr = []

    ✅ C++ Solution¶

    #include <bits/stdc++.h>
using namespace std;

long long dp[1001][1001][2];

long long solve(const vector<int>& arr, int l, int r, int opIndex) {
    if (l > r) return 0;
    int parity = opIndex % 2;
    if (dp[l][r][parity] != LLONG_MIN) return dp[l][r][parity];

    long long total = accumulate(arr.begin() + l, arr.begin() + r + 1, 0LL);
    long long gain = (parity == 1) ? total : -total;

    long long removeLeft = gain + solve(arr, l + 1, r, opIndex + 1);
    long long removeRight = gain + solve(arr, l, r - 1, opIndex + 1);

    return dp[l][r][parity] = max(removeLeft, removeRight);
}

long long getMaximumScore(vector<int> integerArray) {
    int n = integerArray.size();
    for (int i = 0; i <= n; ++i)
        for (int j = 0; j <= n; ++j)
            dp[i][j][0] = dp[i][j][1] = LLONG_MIN;

    return solve(integerArray, 0, n - 1, 1);
}

int main() {
    vector<int> arr = {1, 2, 3, 4, 2, 6};
    cout << getMaximumScore(arr) << endl; // Output: 13

    // Example with large numbers to test overflow handling
    vector<int> bigArr(1000, 1000000000);
    cout << getMaximumScore(bigArr) << endl; // Should not overflow
}
