    Graphs¶

    Graphs consist of nodes (vertices) and edges. They model relationships and connections between entities and are fundamental in computer science for representing networks, maps, and more.

    📖 Key Concepts¶

    • Graph Types: directed / undirected, weighted / unweighted, cyclic / acyclic, DAG
    • Density: sparse (adjacency list) vs dense (matrix)
    • Traversal: BFS (level order, shortest unweighted) vs DFS (structure, cycles, components)
    • Shortest Paths: BFS (unweighted), Dijkstra (non-negative weights), Bellman-Ford (neg edges), Floyd-Warshall (all-pairs), 0-1 BFS (edges of weight 0/1), A* (heuristic)
    • Ordering: Topological sort (Kahn / DFS postorder) for DAGs
    • Connectivity: components, bridges, articulation points, strongly connected components (Kosaraju / Tarjan)
    • Trees vs Graphs: Trees are connected, acyclic, V = E + 1 (undirected)
    • Representation trade-offs: adjacency list (space), matrix (O(1) edge check), edge list (sorting/DSU)
    • Common Patterns: multi-source BFS, layering, coloring, DSU for connectivity/MST, heap for weighted shortest paths

    Core Implementations (Embedded Header)¶

    #pragma once
#include <bits/stdc++.h>

namespace graph {

using AdjList = std::vector<std::vector<int>>;

inline std::vector<int> bfs(int start, const AdjList& adj) {
    std::vector<int> order;
    std::vector<char> seen(adj.size());
    std::queue<int> q;
    q.push(start);
    seen[start] = 1;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : adj[u]) if (!seen[v]) { seen[v] = 1; q.push(v); }
    }
    return order;
}

inline void dfsRec(int u, const AdjList& adj, std::vector<char>& seen, std::vector<int>& order) {
    seen[u] = 1;
    order.push_back(u);
    for (int v : adj[u]) if (!seen[v]) dfsRec(v, adj, seen, order);
}
inline std::vector<int> dfs(int start, const AdjList& adj) {
    std::vector<int> order; std::vector<char> seen(adj.size());
    dfsRec(start, adj, seen, order);
    return order;
}

inline std::vector<int> topo_kahn(const AdjList& adj) {
    int n = (int)adj.size();
    std::vector<int> indeg(n), order;
    for (int u = 0; u < n; ++u) for (int v : adj[u]) indeg[v]++;
    std::queue<int> q; for (int i = 0; i < n; ++i) if (!indeg[i]) q.push(i);
    while (!q.empty()) {
        int u = q.front(); q.pop(); order.push_back(u);
        for (int v : adj[u]) if (--indeg[v] == 0) q.push(v);
    }
    if ((int)order.size() != n) return {};
    return order;
}

struct WeightedEdge { int to; int w; };
using WAdjList = std::vector<std::vector<WeightedEdge>>;

inline std::vector<long long> dijkstra(const WAdjList& g, int src) {
    const long long INF = (1LL<<60);
    int n = (int)g.size();
    std::vector<long long> dist(n, INF);
    dist[src] = 0;
    using P = std::pair<long long,int>;
    std::priority_queue<P, std::vector<P>, std::greater<P>> pq;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [d,u] = pq.top(); pq.pop();
        if (d != dist[u]) continue;
        for (auto [v,w] : g[u]) {
            long long nd = d + w;
            if (nd < dist[v]) { dist[v] = nd; pq.push({nd, v}); }
        }
    }
    return dist;
}

inline void apDfs(int u, int parent, const AdjList& g, std::vector<int>& tin, std::vector<int>& low,
                  std::vector<char>& vis, int& timer, std::vector<char>& isAP) {
    vis[u] = 1; tin[u] = low[u] = ++timer; int children = 0;
    for (int v : g[u]) {
        if (v == parent) continue;
        if (!vis[v]) {
            ++children;
            apDfs(v, u, g, tin, low, vis, timer, isAP);
            low[u] = std::min(low[u], low[v]);
            if (parent != -1 && low[v] >= tin[u]) isAP[u] = 1;
        } else low[u] = std::min(low[u], tin[v]);
    }
    if (parent == -1 && children > 1) isAP[u] = 1;
}
inline std::vector<int> articulation_points(const AdjList& g) {
    int n = (int)g.size();
    std::vector<int> tin(n,-1), low(n,-1);
    std::vector<char> vis(n), isAP(n);
    int timer = 0;
    for (int i = 0; i < n; ++i) if (!vis[i]) apDfs(i, -1, g, tin, low, vis, timer, isAP);
    std::vector<int> res; for (int i = 0; i < n; ++i) if (isAP[i]) res.push_back(i);
    return res;
}

    ⏱ Dijkstra (Min-Heap)¶

    vector<int> dijkstra(int n, vector<vector<pair<int,int>>>& adj, int src) {
    const int INF = 1e9;
    vector<int> dist(n, INF);
    dist[src] = 0;

    using P = pair<int,int>; // (distance, node)
    priority_queue<P, vector<P>, greater<P>> pq;
    pq.push({0, src});

    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d != dist[u]) continue; // stale entry
        for (auto [v, w] : adj[u]) {
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}

    ⚠️ Common Pitfalls¶

    • Re-visiting nodes in Dijkstra when not checking stale heap entries.
    • Using DFS for shortest path (unweighted) instead of BFS -> incorrect ordering.
    • Forgetting to reset visited for multiple component traversals.
    • Confusing directed vs undirected edges in cycle detection.
    • Not handling 0-based vs 1-based node indexing consistently.

    ✅ Interview Checklist¶

    Goal Ask
    Input Form Edge list / adjacency / implicit grid?
    Graph Type Directed? Weighted? Negative weights?
    Output Path length only? Actual path? Count of paths?
    Complexity Is O(V+E) acceptable? Need optimization?
    Edge Cases Disconnected? Self-loops? Parallel edges?

    🧠 Tips & Tricks¶

    • Use BFS for shortest path in unweighted graphs.
    • Always track visited nodes to avoid infinite loops.
    • For grid problems, treat grids as graphs (neighbors: up/down/left/right).
    • Remember: directed vs undirected affects traversal and cycles.

    🔧 Bonus: Articulation Points (Tarjan) in C++¶

    // Finds articulation points in an undirected graph (0..n-1)
void dfsAP(int u, int parent, vector<vector<int>>& adj,
           vector<int>& tin, vector<int>& low, vector<bool>& vis,
           int& timer, vector<bool>& isAP) {
    vis[u] = true;
    tin[u] = low[u] = ++timer;
    int children = 0;
    for (int v : adj[u]) {
        if (v == parent) continue;
        if (!vis[v]) {
            ++children;
            dfsAP(v, u, adj, tin, low, vis, timer, isAP);
            low[u] = min(low[u], low[v]);
            if (parent != -1 && low[v] >= tin[u])
                isAP[u] = true;
        } else {
            low[u] = min(low[u], tin[v]);
        }
    }
    if (parent == -1 && children > 1)
        isAP[u] = true;
}

vector<int> articulationPoints(int n, vector<vector<int>>& adj) {
    vector<int> tin(n, -1), low(n, -1);
    vector<bool> vis(n, false), isAP(n, false);
    int timer = 0;
    for (int i = 0; i < n; ++i)
        if (!vis[i])
            dfsAP(i, -1, adj, tin, low, vis, timer, isAP);
    vector<int> res;
    for (int i = 0; i < n; ++i)
        if (isAP[i]) res.push_back(i);
    return res;
}

    🧱 Alternative: Compact Adjacency List Build¶

    struct Edge { int v, w; };
vector<vector<Edge>> buildWeighted(int n, const vector<tuple<int,int,int>>& edges, bool directed=false) {
    vector<vector<Edge>> g(n);
    for (auto [u,v,w] : edges) {
        g[u].push_back({v,w});
        if (!directed) g[v].push_back({u,w});
    }
    return g;
}
