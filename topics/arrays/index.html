    🛠️ Core Patterns¶

    1. Sliding Window¶

    • Find subarrays or substrings that match certain properties.
    • Problems: Maximum/Minimum subarray sum, longest substring without repeating characters.
    int maxSumSubarray(vector<int>& nums, int k) {
    int n = nums.size();
    int max_sum = 0, curr_sum = 0;
    for (int i = 0; i < k; ++i) curr_sum += nums[i];
    max_sum = curr_sum;
    for (int i = k; i < n; ++i) {
        curr_sum += nums[i] - nums[i - k];
        max_sum = max(max_sum, curr_sum);
    }
    return max_sum;
}

    2. Two Pointers¶

    • Used for searching pairs/triplets, reversing, or partitioning.
    • Problems: Pair sum, move zeroes, reverse array in-place.
    vector<int> twoSumSorted(vector<int>& nums, int target) {
    int left = 0, right = nums.size() - 1;
    while (left < right) {
        int s = nums[left] + nums[right];
        if (s == target) return {left, right};
        else if (s < target) ++left;
        else --right;
    }
    return {};
}

    3. Prefix Sum¶

    • Precompute cumulative sums to answer range queries in O(1).
    • Problems: Subarray sum equals k, range sum query.
    vector<int> prefixSums(vector<int>& nums) {
    vector<int> ps(nums.size() + 1, 0);
    for (int i = 0; i < nums.size(); ++i)
        ps[i + 1] = ps[i] + nums[i];
    return ps;
}

    4. Hashing with Arrays¶

    • Use hash maps (unordered_map/set) to count frequencies, detect duplicates, or map values.
    • Problems: Two sum, longest consecutive sequence.
    bool hasDuplicate(vector<int>& nums) {
    unordered_set<int> seen;
    for (int n : nums) {
        if (seen.count(n)) return true;
        seen.insert(n);
    }
    return false;
}

    5. Binary Search¶

    • For searching in sorted arrays, or to answer "minimum/maximum" problems (search space).
    • Problems: Find element, first/last occurrence, search insert position.
    int binarySearch(vector<int>& nums, int target) {
    int left = 0, right = nums.size() - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (nums[mid] == target) return mid;
        else if (nums[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

    🧠 Tips & Tricks¶

    • Know your STL algorithms: (e.g., std::accumulate, std::min_element, std::max_element)
    • Always check for empty arrays and edge cases (one element, all same, all zeros).
    • Think about space complexity: Can you do it in-place?
    • Know when to use O(n) vs O(n log n) approaches (prefix sum vs sorting).

    📋 Array Template Functions¶

    // Reverse in-place
void reverse(vector<int>& arr) {
    int left = 0, right = arr.size() - 1;
    while (left < right) {
        swap(arr[left], arr[right]);
        ++left;
        --right;
    }
}

// Rotate right by k
void rotate(vector<int>& arr, int k) {
    int n = arr.size();
    k = k % n;
    reverse(arr.begin(), arr.end());
    reverse(arr.begin(), arr.begin() + k);
    reverse(arr.begin() + k, arr.end());
}
